${video.category}

${video.title}

${video.description}

`; card.querySelector(".watch-btn").addEventListener("click", () => { openStream(originalIndex); }); videoGrid.appendChild(card); }); emptyState.classList.toggle("active", filteredVideos.length === 0); }
